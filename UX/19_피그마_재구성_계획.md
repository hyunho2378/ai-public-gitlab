window.__centerAudit = (rootIds) => {
  const out = [];
  const ok = (a, b) => Math.abs(a - b) <= 0.5;
  for (const id of rootIds) {
    const root = figma.getNodeById(id);
    const nodes = [root, ...(root.findAll ? root.findAll(n => n.type === 'FRAME' || n.type === 'COMPONENT' || n.type === 'INSTANCE') : [])];
    for (const n of nodes) {
      if (!n.layoutMode || n.layoutMode === 'NONE') continue;
      if (!/(^|\/)(Button|Icon button|History button|Tab item|Icon|Badge|Count|Step|Avatar|Actions)/i.test(n.name)) continue;
      const kids = n.children.filter(k => k.visible !== false && k.layoutPositioning !== 'ABSOLUTE');
      if (!kids.length) continue;
      const pb = n.absoluteBoundingBox;
      let x0 = 1e9, y0 = 1e9, x1 = -1e9, y1 = -1e9;
      for (const k of kids) { const b = k.absoluteBoundingBox; x0 = Math.min(x0, b.x); y0 = Math.min(y0, b.y); x1 = Math.max(x1, b.x + b.width); y1 = Math.max(y1, b.y + b.height); }
      const dx = (x0 + x1) / 2 - (pb.x + pb.width / 2), dy = (y0 + y1) / 2 - (pb.y + pb.height / 2);
      const row = { id: n.id, name: n.name, w: n.width, h: n.height,
        align: [n.primaryAxisAlignItems, n.counterAxisAlignItems].join('/'),
        padLR: n.paddingLeft - n.paddingRight, padTB: n.paddingTop - n.paddingBottom,
        dx: Math.round(dx * 100) / 100, dy: Math.round(dy * 100) / 100 };
      const bad = row.align !== 'CENTER/CENTER' || row.padLR !== 0 || row.padTB !== 0 || !ok(dx, 0) || !ok(dy, 0) || n.width % 2 || n.height % 2;
      if (bad) out.push(row);
    }
  }
  return out;
};

## 4. 화면 페이지 구성

### 4-1. 섹션과 프레임 배치

- 섹션 하나가 화면 하나. 섹션 이름은 지금 세트 이름 그대로(예: `R-01 Project home`)
- 프레임 이름: `화면번호 / 속성=값, 속성=값`(예: `R-01 / Breakpoint=1440, Auth=Guest, Drawer=Closed, Sidebar=Expanded, Info=Short`). 빗금 뒤는 지금 변형 이름과 글자 하나까지 같게(I-06)

- 행: 첫 행 Auth=Guest 기본, 둘째 행 Auth=Signed-in 기본, 그 아래 추가 상태 한 행씩(Drawer=Open, Dialog=Open, Kind=NotFound, Role 4값, State=Empty와 NoResults, Sidebar=Collapsed, Info=Expanded). 행 사이 160, 행 높이는 그 행에서 가장 높은 프레임
- 빈 칸: 해당 조합 프레임이 없으면 비워 둠(채움용 프레임 금지)
- 행 이름표와 열 이름표: 섹션 안 왼쪽 열과 위쪽 줄에 글자 레이어(`Label / Auth=Guest`, `Label / 1440`), 글자 스타일 연결
- 섹션 안 여백 160, 섹션 사이 320, 페이지 안 섹션은 화면 번호순 2열 격자(10. 30., 11. 1. 배치 규칙)
- 좌표만 정하는 단계와 속성, 연결을 바꾸는 단계를 섞지 않음

### 4-2. 화면 설계서(Annotation) 넣는 방법

1. 주석 분류 만들기: Default value, Data state, Interaction, Responsive, Accessibility(figma.annotations 분류 추가)
2. 섹션마다 설계서 프레임 하나: 이름 `Spec / R-01`, 폭 1920, 섹션 맨 위. export 이름 규칙(빗금 앞 화면번호와 공백)과 겹치지 않는 이름이라 측정에서 빠짐

   - 화면: 번호, 영어 이름, 주소(route), 코드 파일(screens/P*.jsx)
   - 화면 종류: Page, Modal, Sheet 중 하나(M-06)
   - Default value: Guest 첫 화면과 Signed-in 첫 화면, 기본 선택 항목(B-09, M-02)
   - 상태 목록: 프레임 이름과 들어가는 조건 한 줄씩(05_화면_전수목록 문장 사용, M-04)
   - Data state: Min, Max, Empty 기준과 말줄임 규칙(C-01, C-02, M-03)
   - 연결: 이 화면으로 오는 반응 수와 나가는 반응 수(0단계 덤프 값)

## 5. 측정 도구와 프로토타입을 깨지 않는 이관 순서

### 5-1. 이관 원리

- 노드 ID 보존: 변형(COMPONENT) 안 자식을 새 프레임으로 `appendChild`로 옮기면 자식 ID가 그대로 남아 자식에 걸린 반응과 측정 요소 이름이 유지
- 바뀌는 ID: 변형 자체와 세트 자체 둘뿐. 지도 파일 `code_plan/migration/id_map.json`에 `{옛 변형 ID: 새 프레임 ID}`, `{옛 세트 ID: 새 섹션 ID}` 기록
- 반응 재연결: 파일 전체에서 destinationId가 지도에 있는 반응만 새 ID로 바꿔 `setReactionsAsync`. 대상은 화면 프레임 안 노드, Prototype 페이지 노드, Components 페이지 원본 안 노드 전부
- 흐름 시작점: 각 페이지 `flowStartingPoints`의 nodeId도 지도로 교체
- 컴포넌트 속성 참조: 화면 세트에 속성 정의(TEXT, BOOLEAN)가 있으면 자식의 componentPropertyReferences를 옮기기 전에 기록 후 비움. 0단계 값이 0이면 생략
- 변형 자체 값 복사: 크기, fills, strokes(strokeAlign 포함), effects, cornerRadius, clipsContent, 오토 레이아웃 값 전부(layoutMode와 정렬, 패딩, itemSpacing, sizing), layoutGrids, 스타일 ID, boundVariables, explicitVariableModes, exportSettings, 변형 자체 reactions

### 5-2. export_full.js가 읽을 이름 규칙(v2)

- 대상: `type === 'COMPONENT_SET'` 또는 `type === 'SECTION'`이고 이름이 `^[A-Z]-\d+[a-z]?\s`로 시작
